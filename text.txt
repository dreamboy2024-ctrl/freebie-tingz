function isItemOverdue(item) {
if (item.status !== 'Pending' || !item.delivery) return false;
const deliveryDate = new Date(item.delivery);
return deliveryDate < SYSTEM_TODAY;
}
function isItemStalled(item) {
if (item.status !== 'Pending' || !item.requestedDate) return false;
const requestDate = new Date(item.requestedDate);
const diffTime = Math.abs(SYSTEM_TODAY - requestDate);
const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
return diffDays > 14;
}
function getCarrierTrackingUrl(carrier, trackingNum) {
if (!trackingNum) return null;
const cleanNum = trackingNum.trim();
switch(carrier) {
case 'USPS': return https://usps.com{cleanNum};
case 'FedEx': return https://fedex.com{cleanNum};
case 'UPS': return https://ups.com{cleanNum};
case 'DHL': return https://dhl.com{cleanNum};
default: return https://google.com{encodeURIComponent(carrier + ' tracking ' + cleanNum)};
}
}

function toggleStatus(id) {
const index = companies.findIndex(c => c.id === id);
if (index !== -1) {
const current = companies[index].status;
if (current === 'Not Requested') companies[index].status = 'Pending';
else if (current === 'Pending') companies[index].status = 'Received';
else companies[index].status = 'Not Requested';
updateMetrics();
renderList();
}
}
function blacklistCompany(id) {
const index = companies.findIndex(c => c.id === id);
if (index !== -1) {
companies[index].status = 'Blacklisted';
if (!companies[index].notes.includes('[Blacklisted]')) {
companies[index].notes = [Blacklisted] Flagged for verification updates. + companies[index].notes;
}
updateMetrics();
renderList();
}
}
function deleteCompany(id) {
companies = companies.filter(c => c.id !== id);
updateMetrics();
renderList();
}
